cmake_minimum_required(VERSION 3.10)
project(InfernoArm64FramebufferProbe C CXX)

if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This experimental probe requires arm64-v8a")
endif()
set(INFERNO_ROOT "" CACHE PATH "Clean Android ARM64 scalar-profile build")
set(INFERNO_PROBE_JIT 1 CACHE STRING "Explicit interpreter/JIT selection")
if(NOT INFERNO_PROBE_JIT MATCHES "^[01]$")
    message(FATAL_ERROR "INFERNO_PROBE_JIT must be 0 or 1")
endif()
set(corelib "${INFERNO_ROOT}/Android/arm64/lib")
set(core_archives)
foreach(name emu libinterp libmath libkeyring libsec libmp lib9)
    if(NOT EXISTS "${corelib}/${name}.a")
        message(FATAL_ERROR "Missing ARM64 archive: ${corelib}/${name}.a")
    endif()
    list(APPEND core_archives "${corelib}/${name}.a")
endforeach()
add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
set_property(TARGET native_app_glue PROPERTY POSITION_INDEPENDENT_CODE ON)
add_library(inferno_arm64_fb SHARED probe.cpp)
set_target_properties(inferno_arm64_fb PROPERTIES OUTPUT_NAME inferno_os_app)
target_include_directories(inferno_arm64_fb PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
target_compile_features(inferno_arm64_fb PRIVATE cxx_std_11)
target_compile_definitions(inferno_arm64_fb PRIVATE INFERNO_PROBE_JIT=${INFERNO_PROBE_JIT})
set_property(TARGET inferno_arm64_fb APPEND_STRING PROPERTY LINK_FLAGS " -Wl,-u,ANativeActivity_onCreate -Wl,-z,max-page-size=16384")
target_link_libraries(inferno_arm64_fb PRIVATE native_app_glue "-Wl,--start-group" ${core_archives} "-Wl,--end-group" android log dl m)
